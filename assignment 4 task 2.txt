def write_and_append_file():
    # Step 1: Take user input and write it to output.txt
    user_input = input("Enter some text to write to the file: ")
    with open("output.txt", "w") as file:
        file.write(user_input + "\n")

    # Step 2: Take additional input and append it to the file
    more_input = input("Enter additional text to append: ")
    with open("output.txt", "a") as file:
        file.write(more_input + "\n")

    # Step 3: Read and display final content
    print("\nFinal content of output.txt:")
    with open("output.txt", "r") as file:
        print(file.read())

# Run the function
write_and_append_file()
